echo "=== FIRE SIGNAL (правильный формат) ==="
python3 << 'FIRE'
import urllib.request,json
B="http://192.168.3.121:8000"
# Fire signal с правильным форматом
for payload in [
    {"signal_name":"manual"},
    {"signal_name":"manual","context":{"cmd":"id"}},
    {"signal_name":"cxp-test2"},
    {"signal_name":"trigger","content_id":1},
]:
    req=urllib.request.Request(B+"/api/v1/rules/signals/fire",
        data=json.dumps(payload).encode(),
        headers={"Content-Type":"application/json"})
    try:
        r=urllib.request.urlopen(req,timeout=10)
        print(f"FIRE {payload['signal_name']}: {r.status} {r.read()[:300]}")
    except urllib.error.HTTPError as e:
        print(f"FIRE {payload['signal_name']}: {e.code} {e.read()[:200]}")
# Также: PUT rule с Python кодом
rule_code={"name":"cxp-exec","codeType":"python","conditionType":"expression","conditionExpr":"True","actionCode":"import os; result = os.popen('id; hostname; whoami').read()"}
req=urllib.request.Request(B+"/api/v1/rules/3",data=json.dumps(rule_code).encode(),
    headers={"Content-Type":"application/json"},method="PUT")
try:
    r=urllib.request.urlopen(req,timeout=8)
    print(f"RULE-UPDATE: {r.status} {r.read()[:200]}")
except urllib.error.HTTPError as e:
    print(f"RULE-UPDATE: {e.code} {e.read()[:200]}")
# GET rule 3 (что внутри)
req=urllib.request.Request(B+"/api/v1/rules/3")
try:
    r=urllib.request.urlopen(req,timeout=8)
    print(f"RULE-GET: {r.read()[:300]}")
except Exception as e: print(f"RULE-GET: {e}")
FIRE
echo FIRE-SIGNAL
