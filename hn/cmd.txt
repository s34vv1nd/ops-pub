import socket,time
def rcmd(s,cmd,t=1):
    s.sendall((cmd+"\r\n").encode()); time.sleep(t)
    try: return s.recv(4000).decode(errors="replace")
    except: return ""
s=socket.socket();s.settimeout(6);s.connect(("192.168.3.121",6379))
# Все ключи
keys=rcmd(s,"KEYS *",2)
print("KEYS:",keys[:500])
# DBSIZE
print("DBSIZE:",rcmd(s,"DBSIZE"))
# INFO keyspace
print("KEYSPACE:",rcmd(s,"INFO keyspace")[:200])
# Перебор первых ключей
for line in keys.split("\n"):
    if line.startswith('"') and line.count('"')>=2:
        k=line.split('"')[1]
        v=rcmd(s,f"GET {k}",1)
        print(f"KEY[{k}]:",v[:100])
        if len([x for x in keys.split(chr(10)) if x.startswith('"')])>20: break
s.close()
