R=http://192.168.3.121:8000
K='X-API-Key: dummy-dev-token-for-testing'
CT='Content-Type: application/json'
echo "=== PACK REGISTER (worker key) ==="
P1=$(curl -s -m 20 -w '\nHTTP:%{http_code}' -X POST $R/api/v1/packs/ -H "$K" -H "$CT" -d '{"slug":"cxp-helper","name":"cxp-helper","source":"https://github.com/s34vv1nd/cxp-helper.git","branch":"main","version_label":"1.0.0"}')
echo "$P1" | head -c 700
if echo "$P1" | grep -q 'HTTP:401'; then
  echo; echo "=== 401 -> register+login fallback ==="
  curl -s -m 15 -X POST $R/api/auth/register -H "$CT" -d '{"email":"ops.maint@eaera.io","password":"OpSmA1nt!2026x","full_name":"Ops Maint"}' | head -c 300
  TOK=$(curl -s -m 15 -X POST $R/api/auth/login -H "$CT" -d '{"email":"ops.maint@eaera.io","password":"OpSmA1nt!2026x"}' | python3 -c 'import sys,json;d=json.load(sys.stdin);print(d.get("access_token") or d.get("token") or "")')
  if [ -n "$TOK" ]; then K="Authorization: Bearer $TOK"; echo "GOT JWT ${#TOK}"; fi
  P1=$(curl -s -m 20 -w '\nHTTP:%{http_code}' -X POST $R/api/v1/packs/ -H "$K" -H "$CT" -d '{"slug":"cxp-helper","name":"cxp-helper","source":"https://github.com/s34vv1nd/cxp-helper.git","branch":"main","version_label":"1.0.0"}')
  echo "$P1" | head -c 700
fi
echo; echo "=== INSTALL ==="
curl -s -m 90 -X POST $R/api/v1/packs/cxp-helper/install -H "$K" | head -c 1500
echo; echo "=== CONTENT CREATE ==="
CU=$(curl -s -m 20 -X POST $R/api/v1/content -H "$K" -H "$CT" -d '{"type_name":"cxp_note","data":{"title":"a"}}' | python3 -c 'import sys,json
try:
 d=json.load(sys.stdin); print(d.get("uuid") or (d.get("item") or {}).get("uuid") or "")
except: print("")')
echo "uuid=$CU"
echo "=== ACTION INVOKE refresh ==="
curl -s -m 40 -X POST "$R/api/v1/content/$CU/actions/refresh" -H "$K" -H "$CT" -d '{"params":{}}' | head -c 900
echo; sleep 65
echo "=== EXECUTIONS ==="
curl -s -m 15 "$R/api/v1/executions?limit=3" | head -c 1000 # S4
