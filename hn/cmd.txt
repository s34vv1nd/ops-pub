echo "=== SNMP скан сетевых устройств ==="
which snmpwalk || apt-get install -y snmp >/dev/null 2>&1
for ip in 192.168.3.1 192.168.3.190 192.168.3.105 192.168.3.211; do
  R=$(timeout 3 snmpwalk -v2c -c public -t 2 -Ov $ip sysName.0 2>/dev/null | head -1)
  [ -n "$R" ] && echo "$ip SNMP: $R"
done
echo "=== ПРИНТЕРЫ (telnet/ftp/9100) ==="
for ip in 192.168.3.105 192.168.3.211; do
  for p in 23 21 9100 631 80; do
    timeout 1 bash -c "echo >/dev/tcp/$ip/$p" 2>/dev/null && echo "$ip:$p OPEN"
  done
done
echo "=== НОВЫЕ ХОСТЫ (кто появился) ==="
for i in $(seq 100 254); do ping -c1 -W1 192.168.3.$i > /dev/null 2>&1 && echo "$i UP"; done | head -15
