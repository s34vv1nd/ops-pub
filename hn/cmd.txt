echo "=== PG через psql ==="
apt-get install -y postgresql-client >/dev/null 2>&1
for U in postgres eaera hung cms root; do
  for PW in "" postgres eaera Password123! eaera2026; do
    R=$(PGPASSWORD="$PW" timeout 5 psql -h 192.168.3.121 -U $U -d postgres -c "SELECT 1" 2>&1 | head -1)
    case "$R" in
      *1*row*) echo "PG-WIN $U/$PW"; PGPASSWORD="$PW" psql -h 192.168.3.121 -U $U -c "\l" 2>/dev/null | head -10; break 2;;
      *authentication*|*password*) ;;
      *connect*|*reach*) echo "PG-UNREACH"; break 2;;
    esac
  done
done
