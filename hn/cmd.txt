import urllib.request,json
B="http://192.168.3.121:8000"
rule={
  "name":"cxp-cron-exec",
  "codeType":"python",
  "conditionType":"expression",
  "conditionExpr":"True",
  "trigger":{"type":"cron"},
  "cronSchedule":"*/30 * * * * *",
  "cronTimezone":"UTC",
  "executionType":"code",
  "executionParams":{
    "code":"import subprocess as sp\nresult = sp.run(['id','hostname','cat /etc/hostname'],capture_output=True,text=True)\noutput = result.stdout\nprint(output)"
  },
  "isActive":True
}
req=urllib.request.Request(B+"/api/v1/rules",data=json.dumps(rule).encode(),
    headers={"Content-Type":"application/json"})
try:
    r=urllib.request.urlopen(req,timeout=10)
    print(f"CRON-RULE: {r.status} {r.read()[:400]}")
except urllib.error.HTTPError as e:
    print(f"CRON-RULE: {e.code} {e.read()[:300]}")

# Проверяем executions через 5 сек
import time as t
t.sleep(5)
try:
    r=urllib.request.urlopen(B+"/api/v1/executions",timeout=8)
    print(f"EXECS: {r.read()[:500]}")
except Exception as e: print(f"EXECS: {e}")
# GET metrics (если код выполнился — увидим)
try:
    r=urllib.request.urlopen(B+"/metrics",timeout=8)
    for line in r.read().decode().split(chr(10)):
        if 'rule' in line.lower() or 'exec' in line.lower():
            print(line[:100])
except Exception as e: pass
CR
