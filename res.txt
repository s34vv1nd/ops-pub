import asyncio

import httpx

from src.lib.settings import settings


class AuthAPI:
    async def get_token(self) -> str:
        """
        Get an access token for the API with retry logic.
        """
        headers = {"Content-Type": "application/x-www-form-urlencoded"}
        body = {
            "grant_type": "client_credentials",
            "client_id": settings.CRM_CLIENT_ID,
            "client_secret": settings.CRM_CLIENT_SECRET,
            "scope": "/trade_servers/fetch_config",
        }

        last_exception = None
        for attempt in range(3):
            try:
                async with httpx.AsyncClient() as client:
                    response = await client.post(
                        f"{settings.CRM_API_URL}/oauth/token",
                        headers=headers,
                        data=body,
                    )
                response.raise_for_status()
                print(f"Token response: {response.json()}")
                return response.json().get("access_token")
            except (httpx.RequestError, httpx.HTTPStatusError) as e:
                last_exception = e
                print(f"Attempt {attempt + 1} to get token failed: {e}")
                if attempt < 2:
                    await asyncio.sleep(2)

        raise Exception("Failed to get token after 3 attempts") from last_exception
