from typing import Any

import MT5Manager

# Timeout (seconds) for each MT5 API call.
# If a call does not return within this time, the process is killed
# because the MT5 C++ library does not support per-call timeouts
# and a hung call cannot be interrupted from Python.
MT5_CALL_TIMEOUT = 30

# Dedicated executor for MT5 calls so that a hung thread
# does not consume the default asyncio executor.
# max_workers=2 to avoid false timeouts if a second call arrives
# while the first is still running (queued time counts toward timeout).
_mt5_executor = ThreadPoolExecutor(max_workers=2, thread_name_prefix="mt5")


class MT5ManagerWrapper:
    def __init__(self, login, password, server) -> None:
        self.manager = MT5Manager.ManagerAPI()
        logging.info("settings.MT_SERVER: %s, type: %s", server, type(server))
        logging.info("settings.MT_LOGIN: %s, type: %s", login, type(login))
        logging.info(
            "settings.MT_PASSWORD: %s, type: %s",
            password,
            type(password),
        )
        self.server = server
        self.login = login
        self.password = password
        # Callbacks invoked (from the retry thread) after a successful reconnect.
        # Each entry must be a plain callable — use asyncio.run_coroutine_threadsafe
        # inside the callback if you need to schedule async work.
        self._reconnect_callbacks: list = []

    def connect(self) -> bool:
        result = self.manager.Connect(
            self.server,
            self.login,
            self.password,
            MT5Manager.ManagerAPI.EnPumpModes.PUMP_MODE_FULL,
            timeout=10000,
        )
        if result is False:
            error = MT5Manager.LastError()
            print(f"Failed to connect to server: {error}")
